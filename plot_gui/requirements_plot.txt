# requirements_plot.txt
# neomom_plot GUI — minimal dependencies
#
# tkinter is stdlib but must be present in the Python installation:
#   Ubuntu/Debian: sudo apt install python3-tk
#   Windows:       included by default in python.org installer
#
# mpl_toolkits.mplot3d is part of matplotlib — no separate install needed.
# Indirect dependencies (numpy, pillow, pyshp, shapely etc.) are pulled
# in automatically by pip — do not list them here.
#
# To recreate the venv from scratch:
#   python3 -m venv ~/venvs/neomom_plot
#   source ~/venvs/neomom_plot/bin/activate
#   pip install -r requirements_plot.txt
#   pip install pyinstaller
#   deactivate
#
# To regenerate this file accurately from the active venv:
#   pip install pip-chill
#   pip-chill > requirements_plot.txt
# Do NOT use pip freeze — it dumps everything including indirect deps.

matplotlib
numpy
pandas
cartopy
pyproj
